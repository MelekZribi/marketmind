<script>
const chat = document.getElementById('chat');
const form = document.getElementById('form');
const input = document.getElementById('question');
const sendBtn = document.getElementById('send');
const historyPanel = document.getElementById('historyPanel');
const historyToggle = document.getElementById('historyToggle');
const historyList = document.getElementById('historyList');
const newChatBtn = document.getElementById('newChatBtn');

const MESSAGE_ACCUEIL = "Bonjour. Pose-moi une question sur n'importe quel actif (or, petrole, bitcoin, une action...) — j'irai chercher les donnees les plus recentes avant de repondre.";

// currentConversationId = null tant qu'aucun message n'a ete envoye dans cette
// discussion (elle n'existe pas encore en base -> creee au 1er message).
// historique = les derniers echanges, envoyes au backend pour qu'il comprenne
// le contexte ("oui"/"non"/etc.) - separe du contenu affiche a l'ecran.
let currentConversationId = null;
let historique = [];

function addMessage(text, cls) {
  const div = document.createElement('div');
  div.className = 'msg ' + cls;
  div.textContent = text;
  chat.appendChild(div);
  chat.scrollTop = chat.scrollHeight;
  return div;
}

function viderChat() {
  chat.innerHTML = '';
  addMessage(MESSAGE_ACCUEIL, 'bot');
}

async function chargerListeDiscussions() {
  const res = await fetch('/conversations');
  const convs = await res.json();
  historyList.innerHTML = convs.length === 0
    ? '<p style="color:var(--muted); font-size:13px;">Aucune discussion pour le moment.</p>'
    : convs.map(c => `
        <div class="history-item ${c.id === currentConversationId ? 'active' : ''}" data-id="${c.id}">
          <div>
            <div class="q">${c.title}</div>
            <div class="date">${c.date}</div>
          </div>
          <button class="delete-btn" data-delete-id="${c.id}" title="Supprimer">×</button>
        </div>
      `).join('');
}

async function chargerDiscussion(conversationId) {
  const res = await fetch(`/conversations/${conversationId}/messages`);
  if (!res.ok) return;
  const messages = await res.json();

  currentConversationId = conversationId;
  chat.innerHTML = '';
  if (messages.length === 0) {
    addMessage(MESSAGE_ACCUEIL, 'bot');
  } else {
    messages.forEach(m => addMessage(m.content, m.role === 'user' ? 'user' : 'bot'));
  }
  // On reconstruit l'historique JS (limite aux 10 derniers, comme pour une nouvelle discussion)
  historique = messages.slice(-10).map(m => ({ role: m.role, content: m.content }));

  historyPanel.classList.remove('open');
  input.focus();
}

historyToggle.addEventListener('click', async () => {
  historyPanel.classList.toggle('open');
  if (historyPanel.classList.contains('open')) {
    await chargerListeDiscussions();
  }
});

newChatBtn.addEventListener('click', () => {
  currentConversationId = null;
  historique = [];
  viderChat();
  historyPanel.classList.remove('open');
  input.focus();
});

// Delegation d'evenements : clic sur un item -> charge la discussion,
// clic sur le bouton supprimer -> supprime sans charger.
historyList.addEventListener('click', async (e) => {
  const deleteId = e.target.getAttribute('data-delete-id');
  if (deleteId) {
    e.stopPropagation();
    await fetch(`/conversations/${deleteId}`, { method: 'DELETE' });
    if (parseInt(deleteId) === currentConversationId) {
      currentConversationId = null;
      viderChat();
    }
    await chargerListeDiscussions();
    return;
  }
  const item = e.target.closest('.history-item');
  if (item) {
    chargerDiscussion(parseInt(item.getAttribute('data-id')));
  }
});

// Affiche un graphique Chart.js dans une bulle dediee, sous le message texte.
// Trace aussi 2 barrieres horizontales (support/resistance) si fournies, pour
// visualiser clairement les niveaux cles utilises par l'analyse.
function addChart(chartData) {
  const container = document.createElement('div');
  container.className = 'chart-container';
  const canvas = document.createElement('canvas');
  container.appendChild(canvas);
  chat.appendChild(container);
  chat.scrollTop = chat.scrollHeight;

  const datasets = [{
    label: chartData.label,
    data: chartData.values,
    borderColor: '#3DDC97',
    backgroundColor: 'rgba(61, 220, 151, 0.1)',
    fill: true,
    tension: 0.25,
    pointRadius: 0,
    borderWidth: 2,
  }];

  // Barriere de resistance (plafond) : ligne rouge en pointilles
  if (chartData.resistance !== undefined) {
    datasets.push({
      label: 'Resistance ' + chartData.resistance.toFixed(2),
      data: chartData.labels.map(() => chartData.resistance),
      borderColor: '#E5484D',
      borderDash: [6, 4],
      pointRadius: 0,
      borderWidth: 1.5,
      fill: false,
    });
  }
  // Barriere de support (plancher) : ligne bleue en pointilles
  if (chartData.support !== undefined) {
    datasets.push({
      label: 'Support ' + chartData.support.toFixed(2),
      data: chartData.labels.map(() => chartData.support),
      borderColor: '#5B9DF9',
      borderDash: [6, 4],
      pointRadius: 0,
      borderWidth: 1.5,
      fill: false,
    });
  }

  new Chart(canvas, {
    type: 'line',
    data: {
      labels: chartData.labels,
      datasets: datasets
    },
    options: {
      responsive: true,
      plugins: { legend: { labels: { color: '#E6ECF3' } } },
      scales: {
        x: { ticks: { color: '#7C8899', maxTicksLimit: 8 }, grid: { color: '#232B36' } },
        y: { ticks: { color: '#7C8899' }, grid: { color: '#232B36' } },
      }
    }
  });
}

form.addEventListener('submit', async (e) => {
  e.preventDefault();
  const question = input.value.trim();
  if (!question) return;

  addMessage(question, 'user');
  input.value = '';
  sendBtn.disabled = true;
  const loadingEl = addMessage('Recuperation des donnees et analyse en cours...', 'bot loading');

  try {
    const res = await fetch('/ask', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ question, history: historique, conversation_id: currentConversationId })
    });
    const data = await res.json();
    loadingEl.textContent = data.answer;
    loadingEl.classList.remove('loading');

    if (data.chart) {
      addChart(data.chart);
    }
    // Le backend cree la conversation au 1er message et renvoie son id
    currentConversationId = data.conversation_id;

    // On memorise cet echange pour la prochaine question
    historique.push({ role: 'user', content: question });
    historique.push({ role: 'assistant', content: data.answer });
    // On garde seulement les derniers echanges pour ne pas trop alourdir chaque requete
    if (historique.length > 10) historique = historique.slice(-10);
  } catch (err) {
    loadingEl.textContent = "Erreur de connexion au serveur.";
    loadingEl.classList.remove('loading');
  }
  sendBtn.disabled = false;
});
</script>
